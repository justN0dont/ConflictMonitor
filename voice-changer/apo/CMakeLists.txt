cmake_minimum_required(VERSION 3.20)
project(VoiceChangerAPO CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Portable DSP core (identical to the Python reference; see tests/).
add_library(vc_dsp STATIC dsp/spectral_voice.cpp dsp/chain.cpp)
target_include_directories(vc_dsp PUBLIC dsp)
set_target_properties(vc_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(vc_dsp PRIVATE /W4 /fp:precise)
  # Static CRT: audiodg.exe must not depend on a VC++ redistributable being installed.
  set_property(TARGET vc_dsp PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(vc_dsp PRIVATE -Wall -Wextra -O2)
endif()

add_executable(dsp_cli tests/dsp_cli.cpp)
target_link_libraries(dsp_cli PRIVATE vc_dsp)
if(MSVC)
  set_property(TARGET dsp_cli PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

if(WIN32)
  add_library(VoiceChangerAPO SHARED src/apo.cpp src/media_type.cpp src/dllmain.cpp src/control.cpp src/VoiceChangerAPO.def)
  target_link_libraries(VoiceChangerAPO PRIVATE vc_dsp ole32 uuid advapi32 shell32)
  set_target_properties(VoiceChangerAPO PROPERTIES PREFIX "")
  target_compile_definitions(VoiceChangerAPO PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    target_compile_options(VoiceChangerAPO PRIVATE /W4)
    set_property(TARGET VoiceChangerAPO PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_link_options(VoiceChangerAPO PRIVATE -static -static-libgcc -static-libstdc++)
  endif()

  add_executable(apo_harness tests/apo_harness.cpp src/media_type.cpp)
  target_link_libraries(apo_harness PRIVATE ole32 uuid)
  target_compile_definitions(apo_harness PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    set_property(TARGET apo_harness PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_link_options(apo_harness PRIVATE -static -static-libgcc -static-libstdc++ -municode)
  endif()
endif()
